1.2 Simple animation (bouncing ball)

{
    const w = 400, h = 120, r = 20;
    const svg = d3.create("svg").attr("width", w).attr("height", h);
    svg.append("rect")
        .attr("width", w).attr("height", h)
        .attr("rx", 8)
        .attr("fill", "#f0f4f8");

    const ball = svg.append("circle")
        .attr("r", r).attr("cy", h / 2)
        .attr("fill", "#e76f51");

    let x = r, vx = 3;
    while (true) {
        yield svg.node();
        await Promises.delay(16);
        x += vx;
        if (x >= w-r || x <= r) vx *= -1;
        ball.attr("cx", x);
    }
}

1.3 Reactive bar chart with Observable Plot

viewof n = Inputs.range([3, 20], { value: 8, step: 1, label: "Number of bars" })
data = Array.from({ length: n }, (_, i) => ({
    label: `item ${i + 1}`,
    value: Math.round(10 + Math.random() * 90)
}))
Plot.plot({
    marginLeft: 60,
    x: { label: "Value" },
    y: { label: null },
    marks: [
    Plot.barX(data, { x: "value", y: "label", fill: "steelblue", sort: { y: "-x" } }),
    Plot.ruleX([0])
    ]
})
{
    const w = 400, h = 200;
    const svg = d3.create("svg")
        .attr("width", w).attr("height", h)
        .style("cursor", "crosshair")
        .style("background", "#1a1a2e")
        .style("border-radius", "8px");

    const dot = svg.append("circle")
        .attr("r", 12).attr("cx", w/2).attr("cy", h/2)
        .attr("fill", "#e94560").attr("opacity", 0.9);

    const label = svg.append("text")
        .attr("fill", "white").attr("font-size", 12)
        .attr("x", 10).attr("y", 20)
        .text("Move your mouse here");

    svg.on("mousemove", (event) => {
        const [mx, my] = d3.pointer(event);
        dot.attr("cx", mx).attr("cy", my);
        label.text(`x: ${Math.round(mx)} y: ${Math.round(my)}`);
    });

    yield svg.node();
}
