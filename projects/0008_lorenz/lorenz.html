An exploration of the Lorenz Attractor and Makie Animation

The lorenz attractor is a beautiful example of chaotic systems. These are traditionally defined as systems that are deterministic but very small differences in initial conditions lead to vastly different, seemingly unpredictable, outcomes. They are defined by the following differential equations:

\[\begin{align} \frac{dx}{dt} &= \sigma (y - x) \\ \frac{dy}{dt} &= x (\rho - z) - y \\ \frac{dz}{dt} &= x y - \beta z \end{align}\]

Base.@kwdef mutable struct Lorenz
    # parameters
    dt::Float64 = 0.01
    σ::Float64 = 10
    ρ::Float64 = 28
    β::Float64 = 8/3
    # Initial conditions
    x::Float64 = 1
    y::Float64 = 1
    z::Float64 = 1
end
Lorenz
function step!(l::Lorenz)
    # Differential Equations
    dx = l.σ * (l.y - l.x)
    dy = l.x * (l.ρ- l.z) - l.y
    dz = l.x * l.y - l.β * l.z
    # Euler Step
    l.x += l.dt * dx
    l.y += l.dt * dy
    l.z += l.dt * dz
    # Return 3D point
    Point3f(l.x, l.y, l.z)
end

dark_latexfonts = merge(theme_dark(), theme_latexfonts());
function plot_lorenz(; fname="lorenz.mp4", hist_length=200, N_frames=1000, colormap=:viridis)
    attractor = Lorenz()
    points = Observable(CircularBuffer{Point3f}(hist_length))
    colors = Observable(Int[])

    fig = Figure()
    ax = Axis3(
        fig[1,1],
        protrusions=(0, 0, 0, 0),
        viewmode=:fit,
        limits=(-30, 30, -30, 30, 0, 50)
    )
    l = lines!(ax, points, color=colors, colormap=colormap, transparency=true)

    record(fig, fname, 1:N_frames, framerate=30) do frame
        push!(points[], step!(attractor))
        push!(colors[], min(length(points[]), 100))
        l.colorrange = (0, min(length(points[]), 100))
        colors[] = colors[]
        points[] = points[]
        ax.azimuth[] = 1.7pi + 0.3 * sin(2pi * frame / N_frames)
    end
end

Interactive GLMakie Plot

The following code can be used to make the lorenz plot interactive. It adds a run button which pauses/plays the simulation. It also includes a menu to change the colormap used to show the history of the simulation.

menu = Menu(fig, options=["viridis", "inferno", "magma", "plasma"])

on(menu.selection) do cmap 
    l.colormap = cmap
end
notify(menu.selection)

run = Button(fig[2, 1]; label="run", tellwidth=false)
isrunning = Observable(false)
on(run.clicks) do clicks; isrunning[] = !isrunning[]; end
on(run.clicks) do clicks
    @async while isrunning[]
        isopen(fig.scene) || break
        push!(points[], step!(attractor))
        push!(colors[], min(length(points[]), 100))
        l.colorrange = (0, min(length(points[]), 100))
        colors[] = colors[]
        points[] = points[]
        sleep(0.001)
    end
end

Multiple Lorenz Simulations

Of course the main thing people discuss with Lorenz systems is that it is an example of a chaotic system. The general definition for chatoic systems are: > Systems where minute changes in initial conditions lead to vastly different outcomes. To visualize how the chaotic nature of the system the following visualization includes multiple starting conditions to see how the outcomes differ.

Helper function to fade the colormap

function fade_colormap(color::Colorant, n=256)
    r, g, b = red(color), green(color), blue(color)
    colors = [RGBA(r, g, b, α^5) for α in LinRange(0f0, 1f0, n)]
    return ColorScheme(colors)
end

function multi_lorenz(; fname="multi_lorenz.mp4", npoints=300, N_frames=1000, framerate=30)
    eps = 1e-8
    num = 10
    attractors = [
        Lorenz(z=1+(n-1)*eps)
        for n in 1:num
            ]

    all_points = [
        Observable(CircularBuffer{Point3f}(npoints))
        for n in 1:num
            ]
    all_colors = [
        Observable(Float32[])
        for n in 1:num
            ]

    blues = [get(ColorSchemes.Blues, t) for t in LinRange(0.5, 0.8, num)]
    cmaps = [fade_colormap(c) for c in blues]

    fig = Figure(size=(10inch, 7inch), fontsize=18pt)
    ax = Axis3(fig[1,2],
               protrusions=(0, 60, 0, 0),
               viewmode=:fit,
               limits = (-30,30,-30,30,0,50),
               )
    Label(fig[1, :, Top()], "Lorenz Attractor",
          fontsize=30pt,
          color=:white,
          valign=:bottom,
          )
    for (i, points) in enumerate(all_points)
        lines!(ax, points; color=all_colors[i], colormap=cmaps[i], transparency=true)
    end

    eq_layout = fig[1,1] = GridLayout(tellwidth=false, tellheight=false)

    Label(eq_layout[1, 1],
          L"\frac{\mathrm{d}x}{\mathrm{d}t} = \sigma(y - x)",
          fontsize=30pt, color=:white, halign=:left) 
    Label(eq_layout[2, 1],
          L"\frac{\mathrm{d}y}{\mathrm{d}t} = x(\rho - z) - y",
          fontsize=30pt, color=:white, halign=:left)
    Label(eq_layout[3, 1],
          L"\frac{\mathrm{d}z}{\mathrm{d}t} = xy - \beta z",
          fontsize=30pt, color=:white, halign=:left)

    record(fig, fname, 1:N_frames, framerate=framerate) do frame
        for (attractor, points, colors) in zip(attractors, all_points, all_colors)
            push!(points[], step!(attractor))
            n = length(points[])
            colors[] = n <= 1 ? Float32[1f0] : collect(LinRange(0f0, 1f0, n))
            points[] = points[]
        end
        ax.azimuth[] = 1.7pi + 0.3 * sin(2pi * frame / (0.5 * N_frames))
    end
end

function reset!()
    for i in 1:num

        all_points[i][] = all_points[i][]
        all_colors[i][] = Float32[]
        attractors[i] = Lorenz(z=1+(i-1)*eps)
    end
end
